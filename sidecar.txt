**Step 3: Write the Copilot config**
```powershell
$f = "$env:USERPROFILE\.copilot\settings.json"
New-Item -ItemType Directory -Force (Split-Path $f) | Out-Null
if (Test-Path $f) { Copy-Item $f "$f.bak"; $cfg = Get-Content $f -Raw | ConvertFrom-Json } else { $cfg = [pscustomobject]@{} }

$tel = [ordered]@{
  enabled        = $true
  endpoint       = "https://prd.int.api.<...>/<base>"     # no /v1/..., no trailing slash
  protocol       = "http/json"
  serviceName    = "copilot-cli"
  captureContent = $false
  headersProvider = [ordered]@{
    path = $node
    args = @("$pkg\bin\schroders-claude.js", "otel-headers")
    refreshIntervalMs = 1080000
  }
}
$cfg | Add-Member -NotePropertyName telemetry -NotePropertyValue $tel -Force
$cfg | ConvertTo-Json -Depth 10 | Set-Content $f -Encoding UTF8
Get-Content $f
```
